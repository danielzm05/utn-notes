Dado el archivo binario generado en el [[3. Ejercicio finales|ejercicio anterior]], desarrolle un programa  
que genere un archivo ordenado por número de legajo (cada registro debe tener los campos legajo y apellido y nombre) para todos los alumnos que se inscribieron una o más veces. Cada legajo debe ocupar una posición única y predecible en el archivo. El intervalo de los legajos es 80001 a 110000, pueden no presentarse todos los legajos.


> [!success] Para resolver este ejercicio se utilizo lo siguiente:
> - [[Método de Burbujeo]]
> - [[1. Introducción a Archivos|Función para la cantidad de registro]]


```c++
#include <iostream>
#include <cstring>
using namespace std;

struct Boleta{
  char nomYAp[25];
  int legajo;
  int codMateria;
  int dia;
  int mes;
  int anio;
};

int cantReg(FILE* , int);
void burbujeo(Boleta[], int);

int main(){

  FILE* fMat = fopen("MATFINALES.bin", "rb");
  FILE* fNuevo = fopen("INCRITOSORDENADOS.bin", "wb");
  
  int cantInscritos = cantReg(fMat, sizeof(Boleta));
  Boleta inscritos[cantInscritos];
  
  fseek(fMat  , 0, SEEK_SET);
  fread(inscritos, sizeof(Boleta), cantInscritos, fMat);
  
  burbujeo(inscritos, cantInscritos);
  
  for (int i = 0; i < cantInscritos; i++) {
      cout << "Nombre: " << inscritos[i].nomYAp
            << " | Legajo: " << inscritos[i].legajo
            << " | Materia: " << inscritos[i].codMateria
            << " | Fecha: " << inscritos[i].dia << "/" << inscritos[i].mes << "/" << inscritos[i].anio
            << endl;
  }
  
  fclose(fMat);
  fclose(fNuevo);
  
  return 0;
}

int cantReg(FILE* f, int tamStruct){
    long regActual = ftell(f);
    
    fseek(f, 0, SEEK_END);
    
    long tamArch = ftell(f);
    int cant = 0;
    
    if (tamArch > 0 && tamStruct > 0)
        cant = (int)(tamArch / tamStruct);
    fseek(f, regActual, SEEK_SET);
    return cant;
}

void burbujeo(Boleta v[], int n){
  Boleta aux;
  for (int i = 1; i < n; i++)
  {
    for (int j = 1; i <= n-i; i++)
    {
      if(v[j-1].legajo > v[j].legajo){
        aux = v[j-1];
        v[j-1]=v[j];
        v[j]=aux;
      }
    }
  }
}
```
