Se dispone un archivo binario de inscripción de alumnos a exámenes finales  
MAESTROFINALES.DAT y otro con las inscripciones del día de hoy DIAFINALES.DAT,  
ambos ordenados ascendente por código de materia y con el siguiente diseño: 

a. `Nro de legajo` (8 dígitos)  
b. `Código de materia` (6 dígitos)  
c. `ApellidoNombre`(25caract)  

Se pide desarrollar un programa que genere un nuevo archivo de inscripciones a finales  
FINALESACT.DAT resultante del apareo de los dos archivos anteriores, con el mismo orden  
y diseño.


> [!success] Para resolver este ejercicio se utilizo:
> - [[Apareo]]


```c++
#include <iostream>
#include <cstring>
using namespace std;

struct Inscripcion{
  int legajo;
  int codMateria;
  char apeYNom[25];
};

int cantReg(FILE* , int);
void apareo(Inscripcion[], int, Inscripcion[], int, FILE*);

int main(){

  FILE* fInsc = fopen("MAESTROFINALES.dat", "rb");
  FILE* fHoy = fopen("DIAFINALES.dat", "rb");
  FILE* fFin = fopen("FINALESACT.dat", "wb");
  
  int cantInscripciones = cantReg(fInsc, sizeof(Inscripcion));
  Inscripcion inscripciones[cantInscripciones];
  fseek(fInsc  , 0, SEEK_SET);
  fread(inscripciones, sizeof(Inscripcion), cantInscripciones, fInsc);
  fclose(fInsc);
  
  int cantInscHoy = cantReg(fHoy, sizeof(Inscripcion));
  Inscripcion inscripcionesHoy[cantInscHoy];
  fseek(fHoy  , 0, SEEK_SET);
  fread(inscripcionesHoy, sizeof(Inscripcion), cantInscHoy, fHoy);
  fclose(fHoy);
  
  apareo(inscripciones, cantInscripciones, inscripcionesHoy, cantInscHoy, fFin);
  fclose(fFin);
  
  return 0;
}

int cantReg(FILE* f, int tamStruct){
    long regActual = ftell(f);
    fseek(f, 0, SEEK_END);
    long tamArch = ftell(f);
    int cant = 0;
    
    if (tamArch > 0 && tamStruct > 0)
        cant = (int)(tamArch / tamStruct);
    fseek(f, regActual, SEEK_SET);
    return cant;
}

void apareo(Inscripcion v1[], int n, Inscripcion v2[], int m, FILE* fFin){
  int i=0, j=0;
  while (i<n || j<m)
  {
    if(j==m || ( i<n && v1[i].legajo < v2[j].legajo)){
      fwrite(&v1[i], sizeof(Inscripcion), 1, fFin);
      i++;
    }else{
      fwrite(&v2[j], sizeof(Inscripcion), 1, fFin);
      j++;
    }
  }
}
```
