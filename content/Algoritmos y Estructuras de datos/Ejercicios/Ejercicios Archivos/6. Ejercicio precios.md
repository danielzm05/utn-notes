Dado un archivo PRECIOS (desordenado) con los precios de cada articulo, donde cada registro contiene:  
a. Nro. de articulo (5 dígitos)  
b. Descripción del articulo (19 caracteres)  
c. Precio por unidad (real)  
d. Cantidad en stock (5 e)  
e. Nro. de proveedor (4 dígitos)

Desarrollar el programa que imprima el contenido del archivo ordenado por:  
1) Nro. de articulo creciente  
2) Descripción del articulo (alfabético creciente)  
3) Nro. de proveedor creciente y dentro del mismo por Nro. de articulo creciente  


























```c++
#include <iostream>
#include <cstring>
using namespace std;

struct Producto {
    int numArticulo;        
    char descripcion[19];        
    float precio;      
    int cantStock;      
    int numProveedor;      
};

int cantReg(FILE* , int);
void burbujeo(Producto[], int);
void burbujeoDescripcion(Producto[], int);
void imprimir(Producto[], int);

int main(){

  FILE* fProductos = fopen("PRECIOS.dat","rb");
  int cantProductos = cantReg(fProductos, sizeof(Producto));
  Producto productos[cantProductos];
  fseek(fProductos, 0, SEEK_SET);
  fread(productos, sizeof(Producto), cantProductos, fProductos);
  fclose(fProductos);
  cout << "\nOrdenados por numero de articulo: \n" << endl;
  burbujeo(productos, cantProductos);
  imprimir(productos, cantProductos);
  cout << "\nOrdenados alfabeticamente por descripcion: \n" << endl;
  burbujeoDescripcion(productos, cantProductos);
  imprimir(productos, cantProductos);
  return 0;
}

int cantReg(FILE* f, int tamStruct){
    long regActual = ftell(f);
    fseek(f, 0, SEEK_END);
    long tamArch = ftell(f);
    int cant = 0;
    if (tamArch > 0 && tamStruct > 0)
        cant = (int)(tamArch / tamStruct);
    fseek(f, regActual, SEEK_SET);
    return cant;
}

void burbujeo(Producto v[], int n){
  Producto aux;
  for(int i=1; i<n; i++){
    for(int j=0; j<n-i; j++){
      if(v[j].numArticulo > v[j+1].numArticulo){
        aux = v[j];
        v[j] = v[j+1];
        v[j+1] = aux;
      }
    }
  }
}

void imprimir(Producto v[], int n){
  for(int i=0; i<n; i++){
    cout << "Num articulo: " << v[i].numArticulo << ", Descripcion: " << v[i].descripcion
         << ", Precio: " << v[i].precio << ", Stock: " << v[i].cantStock
         << ", Proveedor: " << v[i].numProveedor << endl;
  }
}

void burbujeoDescripcion(Producto v[], int n) {
    Producto aux;
    for (int i = 1; i < n; i++) {
        for (int j = 0; j < n - i; j++) {
            if (strcmp(v[j].descripcion, v[j + 1].descripcion) > 0) {
                aux = v[j];
                v[j] = v[j + 1];
                v[j + 1] = aux;
            }
        }
    }
}
```
