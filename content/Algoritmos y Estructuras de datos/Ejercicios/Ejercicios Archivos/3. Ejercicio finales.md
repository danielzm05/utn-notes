Dado el archivo binario generado en [[2. Ejercicio boletas alumnos|el ejercicio anterior]], desarrolle un programa  
que solicitando por teclado un código de materia permita seleccionar todos los registros que se anotaron para rendirla y los grabe en otro archivo (MATFINALES.DAT), con el mismo  
diseño.

```c++
#include <iostream>
#include <cstring>
using namespace std;

struct Boleta{
  char nomYAp[25];
  int legajo;
  int codMateria;
  int dia;
  int mes;
  int anio;
};

int main(){

  FILE* f = fopen("DIAFINALES.bin", "rb+");
  FILE* fMat = fopen("MATFINALES.bin", "wb+");
  Boleta leer;
  
  int codMateria;
  cout<<"Ingrese un codigo de materia:";
  cin>>codMateria;
  
  while (fread(&leer, sizeof(Boleta), 1, f))
  {
    if (leer.codMateria==codMateria)
    {
      cout << leer.nomYAp<<endl;
      cout<< leer.legajo<<endl;
      cout<< leer.codMateria<<endl;
      cout<< leer.dia<<endl;
      cout<< leer.mes<<endl;
      cout<< leer.anio<< endl;
      cout<<endl;
      fwrite(&leer, sizeof(Boleta), 1, fMat);
    }
  }
  
  fclose(f);
  fclose(fMat);
  
  return 0;
}
```
