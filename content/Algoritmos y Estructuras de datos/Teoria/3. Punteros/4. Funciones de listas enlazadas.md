#### Insertar Ordenado

```c++
Nodo* insertarOrdenado(Nodo*& l, int v) {
    Nodo* nuevo = new Nodo();
    nuevo->info = v;
    nuevo->sgt = NULL;

    Nodo* ant = NULL;
    Nodo* actual = l;

    // Avanzar hasta encontrar posición correcta
    while (actual != NULL && actual->info < v) {
        ant = actual;
        actual = actual->sgt;
    }

    // Insertar al principio
    if (ant == NULL) {
        nuevo->sgt = l;
        l = nuevo;
    }
    else {
        // Insertar en medio o final
        ant->sgt = nuevo;
        nuevo->sgt = actual;
    }

    return nuevo;
}
```

### Eliminar Nodo

Elimina de la lista el valor x, en caso de encontrarlo, retorna verdadero en
esta caso y falso en caso contrario.

```cpp
void eliminarNodo(Nodo*& l, int v){
 Nodo* actual = l;
 Nodo* ant = NULL;
 
 while( actual!=NULL && actual->info!=v ) {
	 ant = actual;
	 actual = actual->sig;
 }
 
 if(actual->info!=v) return; //no lo encontro
 
 if( ant!=NULL )//si no es el primer nodo
 ant->sig = actual->sig;
 
 else//si se elimina el primero
 
 l = actual->sig;//actualiza puntero al inicio
 delete actual;
 
return;
}
```
